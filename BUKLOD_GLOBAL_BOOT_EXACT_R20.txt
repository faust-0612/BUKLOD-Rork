function SwipeDashboardHeader() {
  const pathname =
    usePathname();

  const insets =
    useSafeAreaInsets();

  const { theme } = useBuklodTheme();

  let title = 'Buklod';

  let placement:
    | 'home'
    | 'messages'
    | 'learn'
    | 'community'
    | 'me'
    | null = null;

  if (
    pathname === '/home' ||
    pathname.startsWith('/home/')
  ) {
    title = 'Buklod';
    placement = 'home';
  } else if (
    pathname === '/messages' ||
    pathname.startsWith('/messages/')
  ) {
    title = 'Buklod';
    placement = null;
  } else if (
    pathname === '/learn' ||
    pathname.startsWith('/learn/')
  ) {
    title = 'Buklod';
    placement = 'learn';
  } else if (
    pathname === '/buklod' ||
    pathname.startsWith('/buklod/')
  ) {
    title = 'Buklod';
    placement = 'community';
  } else if (
    pathname === '/prayer' ||
    pathname.startsWith('/prayer/')
  ) {
    title = 'Buklod';
  } else if (
    pathname === '/me' ||
    pathname.startsWith('/me/')
  ) {
    title = 'Buklod';
  }

  return (
    <View
      style={{
        paddingTop: insets.top,
        minHeight: 54 + insets.top,
        backgroundColor: theme.headerSurface,
        flexDirection: 'row',
        alignItems: 'center',
        justifyContent: 'space-between',
        paddingLeft: 16,
      }}
    >
      <Text
        numberOfLines={1}
        style={{
          flex: 1,
          color: theme.accent,
          fontSize: 26,
          fontWeight: '900',
        }}
      >
        {title}
      </Text>

      {placement ? (
        <BuklodHeaderSponsor
          placement={placement}
        />
      ) : (
        <View style={{ width: 12 }} />

    </View>
  );
}

export default function TabsLayout() {
  const { theme } = useBuklodTheme();
  const { activeScope } = useBuklodScope();
  const showPrayer = Boolean(
    activeScope?.show_faith_content
  );

  const {
    session,
    loading,
    membershipState,
    profile,
  } = useAuth();
  const {
    loading: complianceLoading,
    requiresAcceptance,
  } = useCompliance();
  const {
    loading: runtimeLoading,
    config,
  } = useRuntimeConfig();

  useEffect(() => {
    warmBuklodInterstitial();
  }, []);

  if (
    loading ||
    complianceLoading ||
    runtimeLoading ||
    membershipState === 'loading'
  ) {
    return (
      <View
        style={{
          flex: 1,
          alignItems: 'center',
          justifyContent: 'center',
          backgroundColor: theme.canvas,
        }}
      >
        <ActivityIndicator color={theme.accent} />
      </View>
    );
  }
  if (!session) {
    return <Redirect href="/(auth)/login" />;
  }
  if (config?.maintenance_mode) {
    return <Redirect href="/system/maintenance" />;
  }
  if (config?.update_required) {
    return <Redirect href="/system/update-required" />;
  }
  if (
    config?.pilot_access_required &&
    !config?.pilot_access_granted
  ) {
    return <Redirect href="/system/pilot-access" />;
  }
  if (requiresAcceptance) {
    return <Redirect href="/privacy/consent" />;
  }

  const hasCommunityAccess =
    profile?.is_active === true &&
    !!profile?.connection_type;

  if (
    membershipState !== 'ready' &&
    !hasCommunityAccess
  ) {
    return <Redirect href="/onboarding" />;
  }

  if (!profile) {
    return <Redirect href="/onboarding" />;
  }

  const tabSceneStyle = {
    backgroundColor: theme.surface,
    marginHorizontal: 5,
    marginVertical: 3,
    borderRadius: 18,
    overflow: 'hidden' as const,
  };

  /*
   * Instant tab switches (same path as Notifications → Community):
   * Stack + animation:none instead of Material Top Tab ViewPager.
   * Bottom nav is the primary tab control; swipe-between-tabs removed.
   */
  return (
    <View style={{ flex: 1 }}>
      <View
        style={{
          flex: 1,
          backgroundColor: theme.canvas,
        }}
      >
        <SwipeDashboardHeader />
        <BuklodDashboardContextBar />

        <Stack
          screenOptions={{
            headerShown: false,
            animation: 'none',
            freezeOnBlur: true,
            contentStyle: tabSceneStyle,
          }}
        >
          <Stack.Screen name="home" />
          <Stack.Screen name="messages" />
          <Stack.Screen name="buklod" />
          {showPrayer ? (
            <Stack.Screen name="prayer" />
          ) : null}
          <Stack.Screen name="me" />
          <Stack.Screen name="learn" />
        </Stack>
      </View>
    </View>
  );
}

import { Session, User } from '@supabase/supabase-js';
import React, {
  createContext,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
} from 'react';
import { supabase } from '@/lib/supabase';

export type AppRole =
  | 'member'
  | 'leader'
  | 'mentor'
  | 'pastor'
  | 'local_admin'
  | 'system_admin';

export type MembershipState =
  | 'loading'
  | 'ready'
  | 'pending'
  | 'rejected'
  | 'none';

export type Profile = {
  id: string;
  full_name: string;
  first_name?: string | null;
  middle_name?: string | null;
  surname?: string | null;
  suffix?: string | null;
  mobile: string | null;
  sex?: 'male' | 'female' | null;
  username?: string | null;
  login_method?: string | null;
  login_identifier?: string | null;
  avatar_url: string | null;
  church_id: string | null;
  is_verified?: boolean;
  connection_type?: 'local_member'|'invited'|'seeking_church'|'community'|null;
  trust_level?: number|null;

  // Legacy primary role.
  // Do not use this as the sole authorization source.
  role: AppRole;

  is_active: boolean;
};

type AuthContextValue = {
  session: Session | null;
  user: User | null;
  profile: Profile | null;

  // Active simultaneous BUKLOD roles.
  roles: AppRole[];

  loading: boolean;
  membershipState: MembershipState;

  hasRole: (role: AppRole) => boolean;
  hasAnyRole: (rolesToCheck: AppRole[]) => boolean;

  refreshProfile: () => Promise<void>;
  refreshRoles: () => Promise<void>;
  refreshMembership: () => Promise<void>;

  signOut: () => Promise<void>;
};

const VALID_ROLES: AppRole[] = [
  'member',
  'leader',
  'mentor',
  'pastor',
  'local_admin',
  'system_admin',
];

const LEADERSHIP_ROLES: AppRole[] = [
  'leader',
  'mentor',
  'pastor',
  'local_admin',
  'system_admin',
];

const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({
  children,
}: {
  children: React.ReactNode;
}) {
  const [session, setSession] = useState<Session | null>(null);
  const [profile, setProfile] = useState<Profile | null>(null);
  const [roles, setRoles] = useState<AppRole[]>([]);
  const [loading, setLoading] = useState(true);

  const [membershipState, setMembershipState] =
    useState<MembershipState>('loading');

  const bootstrappedRef = useRef(false);
  const sessionUserIdRef = useRef<string | null>(null);

  const loadRoles = async (
    userId?: string,
    legacyRole?: AppRole | null
  ): Promise<AppRole[]> => {
    if (!userId) {
      setRoles([]);
      return [];
    }

    const { data, error } = await supabase
      .from('user_roles')
      .select('role')
      .eq('user_id', userId)
      .eq('is_active', true);

    if (error) {
      console.warn('Role load failed:', error.message);

      // Compatibility fallback only if user_roles could not be read.
      const fallback: AppRole[] =
        legacyRole === 'system_admin'
          ? ['system_admin']
          : ['member'];

      if (
        legacyRole &&
        VALID_ROLES.includes(legacyRole) &&
        !fallback.includes(legacyRole)
      ) {
        fallback.push(legacyRole);
      }

      setRoles(fallback);
      return fallback;
    }

    const nextRoles = Array.from(
      new Set(
        (data ?? [])
          .map((row: any) => row.role as AppRole)
          .filter((role: AppRole) => VALID_ROLES.includes(role))
      )
    );

    // Every active BUKLOD account has Member as its baseline role.
    if (
      !nextRoles.includes('system_admin') &&
      !nextRoles.includes('member')
    ) {
      nextRoles.unshift('member');
    }

    setRoles(nextRoles);

    return nextRoles;
  };

  const loadMembership = async (
    userId?: string,
    _activeRoles: AppRole[] = []
  ) => {
    if (!userId) {
      setMembershipState('none');
      return;
    }

    // Verification is now the universal entry gate.
    // Role and group do not bypass System Admin verification.
    const { data: verification, error: verificationError } =
      await supabase
        .from('profiles')
        .select('is_verified')
        .eq('id', userId)
        .maybeSingle();

    if (verificationError) {
      console.warn(
        'Verification load failed:',
        verificationError.message
      );
      setMembershipState('pending');
      return;
    }

    if (verification?.is_verified === true) {
      setMembershipState('ready');
      return;
    }

    const { data: request } = await supabase
      .from('church_join_requests')
      .select('status')
      .eq('user_id', userId)
      .order('created_at', { ascending: false })
      .limit(1)
      .maybeSingle();

    if (request?.status === 'rejected') {
      setMembershipState('rejected');
      return;
    }

    // Signed-in but not System Admin verified:
    // keep the account on the onboarding/verification screen.
    setMembershipState('pending');
  };

  const loadProfile = async (userId?: string) => {
    if (!userId) {
      setProfile(null);
      setRoles([]);
      setMembershipState('none');
      return;
    }

    const { data, error } = await supabase
      .from('profiles')
      .select(
        'id,full_name,first_name,middle_name,surname,suffix,mobile,sex,username,login_method,login_identifier,avatar_url,church_id,connection_type,trust_level,is_verified,role,is_active'
      )
      .eq('id', userId)
      .single();

    if (error) {
      console.warn('Profile load failed:', error.message);
      setProfile(null);
      setRoles([]);
      setMembershipState('none');
      return;
    }

    const nextProfile = data as Profile;

    setProfile(nextProfile);

    const nextRoles = await loadRoles(
      userId,
      nextProfile.role
    );

    await loadMembership(userId, nextRoles);
  };

  useEffect(() => {
    supabase.auth.getSession().then(async ({ data }) => {
      setSession(data.session);
      sessionUserIdRef.current =
        data.session?.user?.id ?? null;

      await loadProfile(data.session?.user.id);

      bootstrappedRef.current = true;
      setLoading(false);
    });

    const { data: listener } =
      supabase.auth.onAuthStateChange(
        async (_event, nextSession) => {
          const nextUserId =
            nextSession?.user?.id ?? null;

          /* PATCH_171C_SIGNIN_DIRECT_HOME_RACE_FIX
           * Only raise the full-screen gate for a real
           * account switch / fresh login — not every
           * session restore after boot or migration reload.
           */
          if (_event === 'SIGNED_IN') {
            const sameUser =
              bootstrappedRef.current &&
              !!nextUserId &&
              nextUserId === sessionUserIdRef.current;

            if (!sameUser) {
              setLoading(true);
            }
          }

          setSession(nextSession);
          sessionUserIdRef.current = nextUserId;

          await loadProfile(nextSession?.user.id);

          bootstrappedRef.current = true;
          setLoading(false);
        }
      );

    return () => {
      listener.subscription.unsubscribe();
    };
  }, []);

  const value = useMemo<AuthContextValue>(() => {
    const hasRole = (role: AppRole) =>
      roles.includes(role);

    const hasAnyRole = (
      rolesToCheck: AppRole[]
    ) =>
      rolesToCheck.some((role) =>
        roles.includes(role)
      );

    return {
      session,
      user: session?.user ?? null,
      profile,
      roles,
      loading,
      membershipState,

      hasRole,
      hasAnyRole,

      refreshProfile: async () => {
        await loadProfile(session?.user.id);
      },

      refreshRoles: async () => {
        const nextRoles = await loadRoles(
          session?.user.id,
          profile?.role
        );

        await loadMembership(
          session?.user.id,
          nextRoles
        );
      },

      refreshMembership: async () => {
        await loadMembership(
          session?.user.id,
          roles
        );
      },

      signOut: async () => {
        await supabase.auth.signOut();
      },
    };
  }, [
    session,
    profile,
    roles,
    loading,
    membershipState,
  ]);

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const value = useContext(AuthContext);

  if (!value) {
    throw new Error(
      'useAuth must be used inside AuthProvider'
    );
  }

  return value;
}

import AsyncStorage from '@react-native-async-storage/async-storage';
import Constants from 'expo-constants';
import React, { createContext, useContext, useEffect, useMemo, useState } from 'react';

export type RuntimeConfig = {
  environment: string;
  maintenance_mode: boolean;
  maintenance_message?: string | null;
  minimum_supported_version: string;
  recommended_version: string;
  update_required: boolean;
  update_recommended: boolean;
  update_message?: string | null;
  pilot_access_required: boolean;
  pilot_access_granted: boolean;
  admin_bypass: boolean;
  release_title?: string | null;
  release_notes?: string | null;
  features: Record<string, boolean>;
};

type RuntimeContextValue = {
  loading: boolean;
  config: RuntimeConfig | null;
  refresh: () => Promise<void>;
  featureEnabled: (key: string) => boolean;
};

const RuntimeContext = createContext<RuntimeContextValue>({
  loading: true,
  config: null,
  refresh: async () => {},
  featureEnabled: () => true,
});

export function RuntimeConfigProvider({children}:{children:React.ReactNode}){
  const {user}=useAuth();
  const {isConnected,isInternetReachable}=useNetwork();
  const [loading,setLoading]=useState(true);
  const [config,setConfig]=useState<RuntimeConfig|null>(null);

  const online=isConnected && isInternetReachable!==false;
  const version=Constants.expoConfig?.version ?? '0.0.0';
  const cacheKey=user ? `buklod:runtime:${user.id}:${appEnvironment}` : null;

  const refresh=async()=>{
    if(!user){
      setConfig(null);
      setLoading(false);
      return;
    }

    if(!online){
      if(cacheKey){
        const cached=await AsyncStorage.getItem(cacheKey);
        if(cached){
          try{ setConfig(JSON.parse(cached)); }catch{}
        }
      }
      setLoading(false);
      return;
    }

    const {data,error}=await supabase.rpc('get_client_runtime_config',{
      p_environment:appEnvironment,
      p_app_version:version,
    });

    if(error){
      console.warn('BUKLOD runtime config:',error.message);
      setLoading(false);
      return;
    }

    const next=data as RuntimeConfig;
    setConfig(next);
    if(cacheKey)await AsyncStorage.setItem(cacheKey,JSON.stringify(next));
    setLoading(false);
  };

  useEffect(()=>{refresh()},[user?.id,online,appEnvironment]);

  const value=useMemo(()=>({
    loading,
    config,
    refresh,
    featureEnabled:(key:string)=>config?.features?.[key] !== false,
  }),[loading,config]);

  return <RuntimeContext.Provider value={value}>{children}</RuntimeContext.Provider>
}

export function useRuntimeConfig(){
  return useContext(RuntimeContext);
}

import AsyncStorage from '@react-native-async-storage/async-storage';
import React, { createContext, useContext, useEffect, useMemo, useState } from 'react';
import Constants from 'expo-constants';
import { supabase } from '@/lib/supabase';
import { useAuth } from '@/providers/AuthProvider';
import { useNetwork } from '@/providers/NetworkProvider';

export type RequiredPolicy = {
  policy_version_id: string;
  policy_type: 'privacy_notice'|'community_guidelines'|'pastoral_care_notice';
  version: string;
  title: string;
  summary: string;
  acceptance_kind: 'acknowledgement'|'agreement';
  accepted: boolean;
};

type ComplianceValue = {
  loading: boolean;
  policies: RequiredPolicy[];
  requiresAcceptance: boolean;
  refresh: () => Promise<void>;
  acceptAll: () => Promise<{ error?: string }>;
};

const ComplianceContext = createContext<ComplianceValue>({
  loading: true,
  policies: [],
  requiresAcceptance: false,
  refresh: async () => {},
  acceptAll: async () => ({}),
});

export function ComplianceProvider({ children }: { children: React.ReactNode }) {
  const { user } = useAuth();
  const { isConnected, isInternetReachable } = useNetwork();
  const [loading,setLoading]=useState(true);
  const [policies,setPolicies]=useState<RequiredPolicy[]>([]);
  const online = isConnected && isInternetReachable !== false;

  const cacheKey = user ? `buklod:policy-ok:${user.id}` : null;

  const refresh=async()=>{
    if(!user){
      setPolicies([]);
      setLoading(false);
      return;
    }

    if(!online){
      const cached = cacheKey ? await AsyncStorage.getItem(cacheKey) : null;
      setPolicies(cached === 'true' ? [] : [{
        policy_version_id:'offline',
        policy_type:'privacy_notice',
        version:'offline',
        title:'Privacy notice review required',
        summary:'Connect to the internet to review and save required policy acknowledgements.',
        acceptance_kind:'acknowledgement',
        accepted:false,
      }]);
      setLoading(false);
      return;
    }

    const {data,error}=await supabase.rpc('current_required_policies');
    if(error){
      const cached = cacheKey ? await AsyncStorage.getItem(cacheKey) : null;
      setPolicies(cached === 'true' ? [] : []);
      setLoading(false);
      return;
    }

    const rows=(data??[]) as RequiredPolicy[];
    setPolicies(rows);
    const ok=rows.every(p=>p.accepted);
    if(cacheKey) await AsyncStorage.setItem(cacheKey,ok?'true':'false');
    setLoading(false);
  };

  useEffect(()=>{ refresh(); },[user?.id,online]);

  const acceptAll=async()=>{
    if(!user) return {error:'Authentication required.'};
    if(!online) return {error:'Connect to the internet to save your acknowledgement.'};

    const appVersion=Constants.expoConfig?.version ?? null;
    const missing=policies.filter(p=>!p.accepted && p.policy_version_id!=='offline');

    for(const p of missing){
      const {error}=await supabase.rpc('accept_required_policy',{
        p_policy_version_id:p.policy_version_id,
        p_app_version:appVersion,
      });
      if(error) return {error:error.message};
    }

    if(cacheKey) await AsyncStorage.setItem(cacheKey,'true');
    await refresh();
    return {};
  };

  const value=useMemo(()=>({
    loading,
    policies,
    requiresAcceptance:policies.some(p=>!p.accepted),
    refresh,
    acceptAll,
  }),[loading,policies]);

  return <ComplianceContext.Provider value={value}>{children}</ComplianceContext.Provider>;
}

export function useCompliance(){
  return useContext(ComplianceContext);
}

const BUKLOD_FRESH_LOGIN_PUBLIC_INTENT_KEY =

type BuklodScopeContextValue = {

  activeScope: BuklodScope | null;

function pickActiveScope(
  rows: BuklodScope[]
): BuklodScope | null {
  return (
    rows.find(
      (scope) =>
        scope.is_active_scope &&
        scope.scope_type !== 'LOCAL'
      /* PATCH_169G_IGNORE_LOCAL_ACTIVE_SCOPE */
    ) ??

      (scope) =>

    ) ??
    rows[0] ??
    null

export function BuklodScopeProvider({
  children,
}: {
  children: React.ReactNode;
}) {
  const { user } =
    useAuth();

  const [scopes, setScopes] =
    useState<BuklodScope[]>([]);

  const [
    activeScope,
    setActiveScope,
  ] =
    useState<BuklodScope | null>(
      null
    );

  const [loading, setLoading] =
    useState(true);

  const [error, setError] =
    useState<string | null>(null);

  const scopesRef = useRef(scopes);
  scopesRef.current = scopes;

  const refresh =
    useCallback(async () => {
      if (!user?.id) {
        setScopes([]);
        setActiveScope(null);
        setError(null);
        setLoading(false);
        return;
      }

      if (scopesRef.current.length === 0) {
        setLoading(true);
      }

      try {
        /* PATCH_172A_SCOPE_HIERARCHY_ENRICHMENT */
        const baseRows =
          await getMyBuklodScopes();

        let rows = baseRows;

        const scopeIds =
          baseRows
            .map(scope => scope.scope_id)
            .filter(Boolean);

        if (scopeIds.length > 0) {
          const {
            data: hierarchyRows,
            error: hierarchyError,
          } =
            await supabase
              .from('buklod_scopes')
              .select('id,parent_scope_id')
              .in('id', scopeIds);

          if (!hierarchyError) {
            const parentById =
              new Map(
                (hierarchyRows ?? [])
                  .map(row => [
                    String(row.id),
                    row.parent_scope_id
                      ? String(row.parent_scope_id)
                      : null,
                  ] as const)
              );

            rows =
              baseRows.map(scope => ({
                ...scope,
                parent_scope_id:
                  parentById.get(scope.scope_id) ??
                  scope.parent_scope_id ??
                  null,
              }));
          }
          else {
            console.warn(
              'BUKLOD scope hierarchy enrichment:',
              hierarchyError
            );
          }
        }

        /*
         * PATCH_172C_R3_ACTIVE_MEMBER_SCOPE_MERGE
         *
         * Every ACTIVE member of an Organization, child Organization
         * (Sub-organization), or Cluster must receive the actual shared
         * scope selector button. Admin role is NOT required.
         *
         * The canonical RPC remains the primary source. This supplements
         * any ACTIVE managed membership omitted by older RPC logic.
         */
        const {
          data: activeManagedMembershipRows,
          error: activeManagedMembershipError,
        } =
          await supabase
            .from('buklod_scope_memberships')
            .select('scope_id')
            .eq('user_id', user.id)
            .eq(
              'membership_status',
              'ACTIVE'
            );

        if (
          activeManagedMembershipError
        ) {
          throw activeManagedMembershipError;
        }

        const activeManagedScopeIds =
          Array.from(
            new Set(
              (
                activeManagedMembershipRows ??
                []
              )
                .map(row =>
                  row.scope_id
                    ? String(
                        row.scope_id
                      )
                    : ''
                )
                .filter(Boolean)
            )
          );

        if (
          activeManagedScopeIds.length >
          0
        ) {
          const {
            data: activeManagedScopeRows,
            error: activeManagedScopeError,
          } =
            await supabase
              .from('buklod_scopes')
              .select(
                'id,name,scope_type,status,visibility,geo_level,parent_scope_id'
              )
              .in(
                'id',
                activeManagedScopeIds
              )
              .in(
                'scope_type',
                [
                  'ORGANIZATION',
                  'CLUSTER',
                ]
              )
              .in(
                'status',
                [
                  'ACTIVE',
                  'RESTRICTED',
                ]
              );

          if (
            activeManagedScopeError
          ) {
            throw activeManagedScopeError;
          }

          const managedById =
            new Map(
              (
                activeManagedScopeRows ??
                []
              ).map(row => [
                String(row.id),
                row,
              ] as const)
            );

          /*
           * Keep richer RPC data for scopes already returned, but make
           * ACTIVE membership and hierarchy authoritative.
           */
          rows =
            rows.map(scope => {
              const managed =
                managedById.get(
                  scope.scope_id
                );

              if (!managed) {
                return scope;
              }

              return {
                ...scope,
                membership_status:
                  'ACTIVE',
                parent_scope_id:
                  managed
                    .parent_scope_id
                    ? String(
                        managed
                          .parent_scope_id
                      )
                    : null,
              };
            });

          const existingScopeIds =
            new Set(
              rows.map(
                scope =>
                  scope.scope_id
              )
            );

          for (
            const managed of
            managedById.values()
          ) {
            const scopeId =
              String(managed.id);

            if (
              existingScopeIds.has(
                scopeId
              )
            ) {
              continue;
            }

            rows.push({
              scope_id:
                scopeId,
              scope_type:
                managed
                  .scope_type as
                  BuklodScope[
                    'scope_type'
                  ],
              name:
                String(
                  managed.name ??
                    'BUKLOD'
                ),
              scope_status:
                managed.status ??
                null,
              visibility:
                managed.visibility ??
                null,
              geo_level:
                managed.geo_level ??
                null,
              parent_scope_id:
                managed
                  .parent_scope_id
                  ? String(
                      managed
                        .parent_scope_id
                    )
                  : null,
              membership_status:
                'ACTIVE',
              role_keys: [],
              is_admin: false,
              is_active_scope:
                false,
              feature_keys: [],
            } as BuklodScope);

            existingScopeIds.add(
              scopeId
            );
          }
        }
        /*
         * PATCH_171K_R3_FRESH_AUTH_PUBLIC_SCOPE
         *
         * A real login action sets this intent BEFORE Supabase emits
         * SIGNED_IN. Therefore the first scope refresh after login
         * ignores any stale persisted Cluster/Organization active scope
         * and immediately presents PUBLIC locally.
         *
         * App reopen has no fresh-login intent, so its existing root
         * route remains global/personal Connect.
         */
        const freshLoginPublicIntent =
          await AsyncStorage.getItem(
            BUKLOD_FRESH_LOGIN_PUBLIC_INTENT_KEY
          );

        const freshPublicScope =
          freshLoginPublicIntent ===
            '1'
            ? (
                rows.find(
                  scope =>
                    scope.scope_type ===
                    'PUBLIC'
                ) ??
                null
              )
            : null;

        const mzbOrganizationScopeId =
          await resolveMzbOrganizationScopeId();

        rows = rows.map(scope => ({
          ...scope,
          is_mzb_organization: isMzbOrganizationScope(
            {
              scope_id: scope.scope_id,
              scope_type: scope.scope_type,
            },
            mzbOrganizationScopeId
          ),
        }));

        setScopes(rows);

        if (freshPublicScope) {
          setActiveScope(
            freshPublicScope
          );

          await AsyncStorage.removeItem(
            BUKLOD_FRESH_LOGIN_PUBLIC_INTENT_KEY
          );

          /*
           * Persist Public without blocking Home navigation.
           */
          void setBuklodActiveScope(
            freshPublicScope.scope_id
          ).catch(error => {
            console.warn(
              'BUKLOD fresh-login Public scope persistence:',
              error
            );
          });
        }
        else {
          if (
            freshLoginPublicIntent ===
            '1'
          ) {
            await AsyncStorage.removeItem(
              BUKLOD_FRESH_LOGIN_PUBLIC_INTENT_KEY
            );
          }

          setActiveScope(
            pickActiveScope(rows)
          );
        }
        setError(null);
      } catch (nextError: any) {
        console.warn(
          'BUKLOD scope load:',
          nextError
        );

        setScopes([]);
        setActiveScope(null);

        setError(
          String(
            nextError?.message ??
            nextError
          )
        );
      } finally {
        setLoading(false);
      }
    }, [user?.id]);

  useEffect(() => {
    void refresh();
  }, [refresh]);

  const selectScope =
    useCallback(
      async (scopeId: string) => {
        const target =
          scopes.find(
            (scope) =>
              scope.scope_id ===
              scopeId
          );

        if (!target) {
          throw new Error(
            'Scope is not available to this account.'
          );
        }

        if (
          target.scope_type ===
            'SYSTEM'
        ) {
          throw new Error(
            'SYSTEM is not a community scope.'
          );
        }

        /* PATCH_169G_BLOCK_LOCAL_SELECT
         * LOCAL is geography metadata/filtering only.
         * It must never become a separate app dashboard.
         */
        if (
          target.scope_type ===
            'LOCAL'
        ) {
          throw new Error(
            'Location is a Public Community feed filter, not a dashboard.'
          );
        }

        setActiveScope(target);

        setScopes(
          (current) =>
            current.map(
              (scope) => ({
                ...scope,
                is_active_scope:
                  scope.scope_id ===
                  scopeId,
              })
            )
        );

        /*
         * PERF: update local scope immediately for navigation;
         * persist active scope without blocking UI transitions.
         */
        void setBuklodActiveScope(
          scopeId
        ).catch((error) => {
          console.warn(
            'BUKLOD scope persistence:',
            error
          );
        });
      },
      [scopes]
    );

  const featureEnabled =
    useCallback(
      (featureKey: string) =>
        Boolean(
          activeScope
            ?.feature_keys
            ?.includes(
              featureKey
            )
        ),
      [activeScope]
    );

  const value =
    useMemo<BuklodScopeContextValue>(
      () => ({
        scopes,
        activeScope,
        loading,
        error,
        refresh,
        selectScope,
        featureEnabled,
      }),
      [
        scopes,
        activeScope,
        loading,
        error,
        refresh,
        selectScope,
        featureEnabled,
      ]
    );

  return (
    <BuklodScopeContext.Provider
      value={value}
    >
      {children}
    </BuklodScopeContext.Provider>
  );
}

export function useBuklodScope() {
  const value =
    useContext(
      BuklodScopeContext
    );

  if (!value) {
    throw new Error(
      'useBuklodScope must be used inside BuklodScopeProvider'
    );
  }

  return value;
}
