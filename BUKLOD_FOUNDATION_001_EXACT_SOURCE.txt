type IconName = React.ComponentProps<typeof MaterialCommunityIcons>['name'];

function roleLabel(role: AppRole) {
  const labels: Record<AppRole, string> = {
    member: 'Member',
    leader: 'Leader',
    mentor: 'Mentor',
    pastor: 'Pastor',
    local_admin: 'Local Admin',
    system_admin: 'System Admin',
  };

  return labels[role];
}

function DrawerSection({
  title,
  subtitle,
  icon,
  open,
  onToggle,
  children,
}: {
  title: string;
  subtitle?: string;
  icon: IconName;
  open: boolean;
  onToggle: () => void;
  children: ReactNode;
}) {
  return (
    <View style={styles.drawer}>
      <Pressable
        onPress={onToggle}
        style={({ pressed }) => [
          styles.drawerHeader,
          pressed && { opacity: 0.78 },
        ]}
      >
        <View style={styles.drawerIcon}>
          <MaterialCommunityIcons
            name={icon}
            size={18}
            color="#275D45"
          />
        </View>

        <View style={{ flex: 1 }}>
          <Text style={styles.drawerTitle}>{title}</Text>

          {!!subtitle && (
            <Text style={styles.drawerSubtitle}>
              {subtitle}
            </Text>
          )}
        </View>

        <MaterialCommunityIcons
          name={open ? 'chevron-up' : 'chevron-down'}
          size={20}
          color="#68756E"
        />
      </Pressable>

      {open && (
        <View style={styles.drawerBody}>
          {children}
        </View>
      )}
    </View>
  );
}

import { useCallback, useEffect, useState } from 'react';
import {
  Alert,
  FlatList,
  Pressable,
  StyleSheet,
  Switch,
  Text,
  View,
} from 'react-native';
import { MaterialCommunityIcons } from '@expo/vector-icons';

import { AdminGate } from '@/components/AdminGate';
import { shared } from '@/components/UI';
import { supabase } from '@/lib/supabase';
import { AppRole, useAuth } from '@/providers/AuthProvider';
import AdminDeleteAccountButton from '@/components/AdminDeleteAccountButton';

const ROLES: AppRole[] = [
  'system_admin',
];

const ROLE_LABELS: Record<AppRole, string> = {
  member: 'Member',
  leader: 'Leader',
  mentor: 'Mentor',
  pastor: 'Pastor',
  local_admin: 'Local Admin',
  system_admin: 'System Admin',
};

const ROLE_DESCRIPTIONS: Record<AppRole, string> = {
  member: 'Baseline BUKLOD membership',
  leader: 'Cluster and member-care leadership',
  mentor: 'Mentoring and teaching access',
  pastor: 'Pastoral and formation validation access',
  local_admin: 'Local church administrative access',
  system_admin: 'System-wide administrative authority',
};

type UserItem = {
  id: string;
  full_name: string | null;
  mobile: string | null;
  role: AppRole;
  is_active: boolean;
  church?: {
    name?: string | null;
  } | null;
  roles: AppRole[];
};

export default function UsersScreen() {
  const {
    profile,
    refreshRoles,
  } = useAuth();

  const [items, setItems] = useState<UserItem[]>([]);
  const [busyKey, setBusyKey] = useState<string | null>(null);

  const load = useCallback(async () => {
    const [
      { data: profiles, error: profileError },
      { data: userRoles, error: rolesError },
    ] = await Promise.all([
      supabase
        .from('profiles')
        .select(
          'id,full_name,mobile,role,is_active,church:church_id(name)'
        )
        .order('full_name'),

      supabase
        .from('user_roles')
        .select('user_id,role,is_active')
        .eq('is_active', true),
    ]);

    if (profileError) {
      Alert.alert('Users', profileError.message);
      return;
    }

    if (rolesError) {
      Alert.alert('Roles', rolesError.message);
      return;
    }

    const roleMap = new Map<string, AppRole[]>();

    for (const row of userRoles ?? []) {
      const role = row.role as AppRole;
      const current = roleMap.get(row.user_id) ?? [];

      if (!current.includes(role)) {
        current.push(role);
      }

      roleMap.set(row.user_id, current);
    }

    const merged: UserItem[] = (profiles ?? []).map(
      (user: any) => {
        const activeRoles =
          roleMap.get(user.id) ?? [];

        if (
          !activeRoles.includes('system_admin') &&
          !activeRoles.includes('member')
        ) {
          activeRoles.unshift('member');
        }

        return {
          ...user,
          roles: activeRoles,
        };
      }
    );

    setItems(merged);
  }, []);

  useEffect(() => {
    load();
  }, [load]);

  const setRole = async (
    user: UserItem,
    role: AppRole,
    active: boolean
  ) => {
    if (role === 'member' && !active) {
      Alert.alert(
        'Member role',
        'Member is the baseline role of every active BUKLOD account.'
      );
      return;
    }

    if (
      user.id === profile?.id &&
      role === 'system_admin' &&
      !active
    ) {
      Alert.alert(
        'Protected account',
        'You cannot remove your own System Admin role.'
      );
      return;
    }

    const key = `${user.id}:${role}`;
    setBusyKey(key);

    const { error } = await supabase.rpc(
      'admin_set_user_role',
      {
        p_user_id: user.id,
        p_role: role,
        p_active: active,
      }
    );

    setBusyKey(null);

    if (error) {
      Alert.alert(
        'Role update failed',
        error.message
      );
      return;
    }

    await load();

    if (user.id === profile?.id) {
      await refreshRoles();
    }
  };

  const setActive = async (
    user: UserItem,
    value: boolean
  ) => {
    const { error } = await supabase.rpc(
      'set_user_active',
      {
        p_user_id: user.id,
        p_active: value,
      }
    );

    if (error) {
      Alert.alert(
        'Status change failed',
        error.message
      );
      return;
    }

    await load();
  };

  return (
    <AdminGate>
      <FlatList
        style={shared.screen}
        contentContainerStyle={styles.content}
        data={items}
        keyExtractor={(item) => item.id}
        showsVerticalScrollIndicator={false}
        ListHeaderComponent={
          <View style={styles.header}>
            <Text style={styles.eyebrow}>
              PEOPLE & ACCESS
            </Text>

            <Text style={styles.title}>
              Users & Roles
            </Text>

            <Text style={styles.subtitle}>
              One account may hold multiple active roles at the same time.
              Roles are independent and administrator-controlled.
            </Text>

            <View style={styles.infoBox}>
              <MaterialCommunityIcons
                name="shield-account-outline"
                size={22}
                color="#275D45"
              />

              <Text style={styles.infoText}>
                App-wide role control now uses System Admin only.
                Organization, Church and Cluster roles are managed
                inside their own scopes.
              </Text>
            </View>
          </View>
        }
        renderItem={({ item }) => (
          <View style={styles.userCard}>
            <View style={styles.userTop}>
              <View style={{ flex: 1 }}>
                <Text style={styles.userName}>
                  {item.full_name || 'Unnamed Member'}
                </Text>

                <Text style={styles.meta}>
                  {item.church?.name || 'No approved church'}
                  {'  |  '}
                  {item.mobile || 'No mobile'}
                </Text>
              </View>

              <View style={styles.statusBox}>

                      <View style={styles.lockRow}>
                        <MaterialCommunityIcons
                          name="lock-outline"
                          size={13}
                          color="#7A867F"
                        />

                          {role === 'member'
                            ? 'Baseline role'
                            : 'Protected'}
                        </Text>
                      </View>
                    )}
                  </Pressable>
                );
              })}
            </View>

            <View style={styles.summary}>
              <Text style={styles.summaryLabel}>
                CURRENT ACCESS
              </Text>

              <Text style={styles.summaryText}>
                {item.roles
                  .map(
                    (role) =>
                      ROLE_LABELS[role]
                  )
                  .join('  |  ')}
              </Text>
            </View>
          

            <AdminDeleteAccountButton
              userId={item.id}
              userName={
                item.full_name ||
                'Unnamed Member'
              }
              disabled={
                busyKey !== null
              }
              onDeleted={() =>
                void load()
              }
            />
</View>
        )}
        ListFooterComponent={
          <View style={{ height: 70 }} />
        }
      />
    </AdminGate>
  );
}

create table if not exists public.user_roles (
  user_id uuid not null
    references public.profiles(id)
    on delete cascade,

  role text not null
    check (
      role in (
        'member',
        'leader',
        'mentor',
        'pastor',
        'local_admin',
        'system_admin'
      )
    ),

  is_active boolean not null default true,

  granted_by uuid
    references public.profiles(id)
    on delete set null,

  granted_at timestamptz not null default now(),

  revoked_by uuid
    references public.profiles(id)
    on delete set null,

  revoked_at timestamptz,

  primary key (user_id, role)
);

user_roles_active_idx
on public.user_roles(user_id, role)
where is_active = true;

  user_id,
  role,
  is_active,
  granted_at
)
select
  p.id,
  'member',
  true,
  now()
from public.profiles p
where p.is_active = true
on conflict (user_id, role)

  is_active = true,

create or replace function public.admin_set_user_role(
  p_user_id uuid,
  p_role text,
  p_active boolean
)
returns void
language plpgsql
security definer
set search_path = ''
as $$
declare
  actor uuid := (select auth.uid());
begin

  if actor is null then
    raise exception 'Authentication required';
  end if;

  if not public.is_system_admin() then
    raise exception 'System Admin access required';
  end if;

  if p_role not in (
    'member',
    'leader',
    'mentor',
    'pastor',
    'local_admin',
    'system_admin'
  ) then
    raise exception 'Invalid BUKLOD role';
  end if;


  -- Every active BUKLOD account remains a member.
  if p_role = 'member' and p_active = false then
    raise exception
      'Member is the baseline role of an active BUKLOD account';
  end if;


  -- Prevent accidental removal of own System Admin access.
  if p_user_id = actor
     and p_role = 'system_admin'
     and p_active = false then
    raise exception
      'You cannot remove your own System Admin role';
  end if;


  insert into public.user_roles (
    user_id,
    role,
    is_active,
    granted_by,
    granted_at,
    revoked_by,
    revoked_at
  )
  values (
    p_user_id,
    p_role,
    p_active,
    case when p_active then actor else null end,
    case when p_active then now() else now() end,
    case when p_active then null else actor end,
    case when p_active then null else now() end
  )

  on conflict (user_id, role)
  do update set
    is_active = excluded.is_active,

    granted_by =
      case
        when excluded.is_active
        then actor
        else public.user_roles.granted_by
      end,

    granted_at =
      case
        when excluded.is_active
        then now()
        else public.user_roles.granted_at
      end,

    revoked_by =
      case
        when excluded.is_active
        then null
        else actor
      end,

    revoked_at =
      case
        when excluded.is_active
        then null
        else now()
      end;


  -- Keep legacy capability records synchronized
  -- during the transition.
  if p_role in ('mentor','pastor') then

    insert into public.user_capabilities (
      user_id,
      capability,
      granted_by,
      granted_at,
      is_active
    )
    values (
      p_user_id,
      p_role,
      actor,
      now(),
      p_active
    )

    on conflict (user_id, capability)
    do update set
      is_active = excluded.is_active,
      granted_by = actor,
      granted_at =
        case
          when excluded.is_active
          then now()
          else public.user_capabilities.granted_at
        end;

  end if;

end;
$$;
revoke all on function
public.admin_set_user_role(uuid,text,boolean)
from public;
grant execute on function
public.admin_set_user_role(uuid,text,boolean)
to authenticated;
