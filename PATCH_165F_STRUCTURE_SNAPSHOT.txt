export default function TabsLayout() {
  const insets = useSafeAreaInsets();

  /* PATCH_154N_SWIPE_TABS */
  const pathname =
    usePathname();

  const swipeResponder =
    useMemo(
      () =>
        PanResponder.create({
          onStartShouldSetPanResponder:
            () => false,

          onMoveShouldSetPanResponder:
            (_, gesture) => {
              const horizontal =
                Math.abs(
                  gesture.dx
                );

              const vertical =
                Math.abs(
                  gesture.dy
                );

              return (
                horizontal > 38 &&
                horizontal >
                  vertical * 1.55
              );
            },

          onPanResponderTerminationRequest:
            () => true,

          onShouldBlockNativeResponder:
            () => false,

          onPanResponderRelease:
            (_, gesture) => {
              const distance =
                Math.abs(
                  gesture.dx
                );

              const velocity =
                Math.abs(
                  gesture.vx
                );

              if (
                distance < 72 &&
                velocity < 0.55
              ) {
                return;
              }

              const routes = [
                '/home',
                '/messages',
                '/learn',
                '/buklod',
                '/prayer',
                '/me',
              ];

              const current =
                routes.indexOf(
                  pathname
                );

              if (current < 0) {
                return;
              }

              const direction =
                gesture.dx < 0
                  ? 1
                  : -1;

              const nextIndex =
                current +
                direction;

              if (
                nextIndex < 0 ||
                nextIndex >=
                  routes.length
              ) {
                return;
              }

              router.navigate(
                routes[
                  nextIndex
                ] as any
              );
            },
        }),
      [pathname]
    );

  const { session, loading, profile } = useAuth();
  const { loading: complianceLoading, requiresAcceptance } = useCompliance();
  const { loading: runtimeLoading, config } = useRuntimeConfig();

  if (loading || complianceLoading || runtimeLoading) {
    return <View style={{flex:1,alignItems:'center',justifyContent:'center'}}><ActivityIndicator color={colors.green}/></View>;
  }
  if (!session) return <Redirect href="/(auth)/login" />;
  if (config?.maintenance_mode) return <Redirect href="/system/maintenance" />;
  if (config?.update_required) return <Redirect href="/system/update-required" />;
  if (config?.pilot_access_required && !config?.pilot_access_granted) return <Redirect href="/system/pilot-access" />;
  if (requiresAcceptance) return <Redirect href="/privacy/consent" />;
  /* PATCH_164A_UI_UNIVERSAL_ACCESS */
  if (!profile) {
    return <Redirect href="/onboarding" />;
  }

  return (
    <View
      style={{ flex: 1 }}
      {...swipeResponder.panHandlers}
    >
      <View
      style={{
        flex:1,
        backgroundColor:
          colors.white,
      }}
    >

      <SwipeDashboardHeader />
      <BuklodTabAdStrip />

      <SwipeTabs
      tabBarPosition="bottom"
      swipeEnabled={true}
      lazy={false}
      keyboardDismissMode="on-drag"
      initialLayout={{
        width:
          Dimensions
            .get('window')

      }}
      screenOptions={{

        },

function homeNewsAge(
  value:
    string |
    null
) {

  if (!value) {
    return 'Latest update';
  }


  const time =
    new Date(
      value
    ).getTime();


  if (
    !Number.isFinite(
      time
    )
  ) {
    return 'Latest update';
  }


  const diff =
    Math.max(
      0,
      Date.now() -
      time
    );


  const minutes =
    Math.floor(
      diff /
      60000
    );


  if (
    minutes <
    60
  ) {

    return minutes <= 1
      ? 'Just now'
      : minutes + ' min ago';

  }


  const hours =
    Math.floor(
      minutes /
      60
    );


  if (
    hours <
    24
  ) {

    return hours === 1
      ? '1 hour ago'
      : hours + ' hours ago';

  }


  const days =
    Math.floor(
      hours /
      24
    );


  if (
    days <
    7
  ) {

    return days === 1
      ? 'Yesterday'
      : days + ' days ago';

  }


  return new Date(
    value
  ).toLocaleDateString();

}

      '&'
    )
    .replace(
      /&lt;/gi,
      '<'
    )
    .replace(
      /&gt;/gi,
      '>'
    )
    .replace(
      /\s+/g,
      ' '
    )
    .trim();

}


function capitalizeFirst(
  value:
    string
) {

  if (!value) {
    return value;
  }


  return (
    value
      .charAt(0)
      .toUpperCase() +
    value.slice(1)
  );

}


/*
 * This does NOT touch the original English.
 *
 * These are presentation rules only for the
 * locally translated Tagalog copy.
 */
function reconstructChurchTagalog(
  input:
    string
) {

  let text =
    cleanTranslatedNewsText(
      input
    );


  /*
   * Church terminology.
   */
  text =
    text
      .replace(
        /\bmga simbahan\b/gi,
        'mga iglesia'
      )
      .replace(
        /\bsimbahan\b/gi,
        'iglesia'
      )
      .replace(
        /\bmga bautista\b/gi,
        'mga Baptist'
      )
      .replace(
        /\bbautista\b/gi,
        'Baptist'
      )
      .replace(
        /\bebanghelyo\b/gi,
        'Ebanghelyo'
      )
      .replace(
        /\bmga misyonero\b/gi,
        'mga misyonero'
      )
      .replace(
        /\bmisyonero\b/gi,
        'misyonero'
      )
      .replace(
        /\bmga misyon\b/gi,
        'mga gawaing misyon'
      )
      .replace(
        /\bministeryo\b/gi,
        'ministeryo'
      )
      .replace(
        /\bpastor\b/gi,
        'pastor'
      );


  /*
   * Make common machine-translated
   * "Ang X ay nag..." sentences sound
   * more naturally Filipino:
   *
   * Ang mga iglesia ay naglunsad...
   *      \u2193
   * Naglunsad ang mga iglesia...
   */
  const angNag =
    text.match(
      /^Ang\s+(.+?)\s+ay\s+(nag[A-Za-z\u00C0-\u00D6\u00D8-\u00F6\u00F8-\u00FF-]+)(.*)$/i
    );


  if (
    angNag
  ) {

    text =
      capitalizeFirst(
        angNag[2]
      ) +
      ' ang ' +
      angNag[1] +
      angNag[3];

  }


  /*
   * Si Juan ay nagsabi...
   *      \u2193
   * Nagsabi si Juan...
   */
  const siNag =
    text.match(
      /^Si\s+(.+?)\s+ay\s+(nag[A-Za-z\u00C0-\u00D6\u00D8-\u00F6\u00F8-\u00FF-]+)(.*)$/i
    );


  if (
    siNag
  ) {

    text =
      capitalizeFirst(
        siNag[2]
      ) +
      ' si ' +
      siNag[1] +
      siNag[3];

  }


  /*
   * Clean common mechanical phrasing.
   */
  text =
    text
      .replace(
        /\bsa pamamagitan ng kanilang mga pagsisikap\b/gi,
        'sa kanilang mga gawain'
      )
      .replace(
        /\bsa komunidad ng simbahan\b/gi,
        'sa komunidad ng iglesia'
      )
      .replace(
        /\bKristiyanong ministeryo\b/gi,
        'Christian ministry'
      )
      .replace(
        /\s+([,.!?;:])/g,
        '$1'
      )
      .replace(
        /\s+/g,
        ' '
      )
      .trim();


  return text;

}


/*
 * MyMemory accepts short translation segments.
 * We keep each request comfortably below the
 * service's maximum by using ~350-char chunks.
 */
function splitNewsForTranslation(
  input:
    string
) {

  const text =
    String(
      input ??
      ''
    )
      .replace(
        /\s+/g,
        ' '
      )
      .trim();


  if (!text) {
    return [];
  }


  const sentences =
    text.match(
      /[^.!?]+[.!?]+|[^.!?]+$/g
    ) ??
    [
      text,
    ];


  const chunks:
    string[] = [];


  for (
    const sentence
    of sentences
  ) {

    const clean =
      sentence.trim();


    if (
      clean.length <=
      350
    ) {

      chunks.push(
        clean
      );

      continue;

    }


    const words =
      clean.split(
        /\s+/
      );


    let current =
      '';


    for (
      const word
      of words
    ) {

      const candidate =
        current
          ? current +
            ' ' +
            word
          : word;


      if (
        candidate.length >
        350
      ) {

        if (
          current
        ) {

          chunks.push(
            current
          );

        }


        current =
          word;

      }
      else {

        current =
          candidate;

      }

    }


    if (
      current
    ) {

      chunks.push(
        current
      );

    }

  }


  return chunks;

}


async function translateNewsSegmentToTagalog(
  segment:
    string
) {

  const url =
    'https://api.mymemory.translated.net/get' +
    '?q=' +
    encodeURIComponent(
      segment
    ) +
    '&langpair=en%7Ctl' +
    '&mt=1';


  const response =
    await fetch(
      url,
      {
        method:
          'GET',

        headers: {
          Accept:
            'application/json',
        },
      }
    );


  if (
    !response.ok
  ) {

    throw new Error(
      'Translation HTTP ' +
      response.status
    );

  }


  const payload =
    await response.json();


  const translated =
    payload
      ?.responseData
      ?.translatedText;


  if (
    typeof translated !==
      'string' ||
    !translated.trim()
  ) {

    const details =
      payload
        ?.responseDetails;


    throw new Error(
      typeof details ===
        'string'
        ? details
        : 'Translation returned no text.'
    );

  }


  return cleanTranslatedNewsText(
    translated
  );

}


async function translateNewsToTagalog(
  input:
    string
) {

  const cleanInput =
    sanitizeHomeNewsText(
      input
    );


  if (
    !cleanInput
  ) {

    return '';

  }


  const chunks =
    splitNewsForTranslation(
      cleanInput
    );


  if (
    chunks.length ===
    0
  ) {

    return '';

  }


  const translated:
    string[] = [];


  /*
   * Sequential requests are intentional.
   * We do not flood the free service.
   */
  for (
    const chunk
    of chunks
  ) {

    const result =
      await translateNewsSegmentToTagalog(
        chunk
      );


    translated.push(
      result
    );

  }


  return reconstructChurchTagalog(
    translated.join(
      ' '
    )
  );

}


async function readHomeNewsTagalogCache() {

  try {

    const raw =
      await AsyncStorage
        .getItem(
          HOME_NEWS_TAGALOG_CACHE_KEY
        );


    if (!raw) {
      return {};
    }


    const parsed =
      JSON.parse(
        raw
      );


    if (
      !parsed ||
      typeof parsed !==
        'object'
    ) {

      return {};

    }


    return parsed as
      LocalTagalogNewsCache;

  }
  catch {

    return {};

  }

}


async function saveHomeNewsTagalogCache(
  cache:
    LocalTagalogNewsCache
) {

  try {

    /*
     * Keep the local cache reasonably small.
     */
    const entries =
      Object.entries(
        cache
      )
        .sort(
          (
            a,
            b
          ) =>
            String(
              b[1]
                ?.translated_at ??
              ''
            ).localeCompare(
              String(
                a[1]
                  ?.translated_at ??
                ''
              )
            )
        )
        .slice(
          0,
          100
        );


    await AsyncStorage
      .setItem(
        HOME_NEWS_TAGALOG_CACHE_KEY,
        JSON.stringify(
          Object.fromEntries(
            entries
          )
        )
      );

  }
  catch {
    // Translation still displays during this session.
  }

}

/* HOME_NEWS_CONTENT_SANITIZER_V1 */

function sanitizeHomeNewsText(
  value:
    string | null | undefined
) {

  let text =
    String(
      value ?? ''
    )
      .replace(
        /\s+/g,
        ' '
      )
      .trim();


  /*
   * -------------------------------------------------------
   * 1. READER METADATA
   * -------------------------------------------------------
   */
  text =
    text.replace(
      /\b(?:Title|URL\s*Source|Published\s*Time|Markdown\s*Content|Markdown|Content|Nilalaman|Source\s*URL|Web\s*Address|Website)\s*:?\s*/gi,
      ' '
    );


  /*
   * -------------------------------------------------------
   * 2. NORMAL RAW URLs
   * -------------------------------------------------------
   */
  text =
    text.replace(
      /https?:\/\/[^\s<>"']+/gi,
      ' '
    );


  text =
    text.replace(
      /www\.[^\s<>"']+/gi,
      ' '
    );


  /*
   * -------------------------------------------------------
   * 3. MANGLED / TRANSLATED URLS
   *
   * Examples:
   *
   * com/resource - library/news/article-name
   * baptistpress.com / resource - library / news
   * org/stories/something
   * -------------------------------------------------------
   */
  text =
    text.replace(
      /\b(?:[a-z0-9-]+\.)?(?:com|org|net|news|co|uk|ph)\s*(?:\/|\\)\s*(?:[a-z0-9_-]+\s*(?:[-/]\s*)?){1,30}/gi,
      ' '
    );


  text =
    text.replace(
      /\b[a-z0-9.-]+\.(?:com|org|net|news|co|uk|ph)(?:\s*[/\\-]\s*[a-z0-9._-]+)+/gi,
      ' '
    );


  /*
   * -------------------------------------------------------
   * 4. IF A REAL NEWS DATELINE EXISTS,
   *    DISCARD EVERYTHING BEFORE IT.
   *
   * Example:
   * junk junk BOSTON (BP) \u2013 actual story...
   * -------------------------------------------------------
   */
  const dateline =
    text.match(
      /\b[A-Z][A-Z .,'\u2019/-]{2,45}\s*\([A-Z0-9]{2,8}\)\s*[\u2014\u2013-]\s*/
    );


  if (
    dateline &&
    typeof dateline.index ===
      'number'
  ) {

    text =
      text.slice(
        dateline.index +
        dateline[0].length
      );

  }


  /*
   * Dateline without parenthetical source:
   * MANILA, Philippines \u2014 actual story...
   */
  text =
    text.replace(
      /^[A-Z][A-Za-z\u00C0-\u00FF .,'\u2019/-]{2,55}\s*[\u2014\u2013-]\s*/,
      ''
    );


  /*
   * -------------------------------------------------------
   * 5. REMOVE LEFTOVER MARKDOWN / PATH GARBAGE
   * -------------------------------------------------------
   */
  text =
    text
      .replace(
        /\[([^\]]+)\]\([^)]*\)/g,
        '$1'
      )
      .replace(
        /\b(?:resource|library|news|article|stories)\s*[/\\-]\s*(?:[a-z0-9_-]+\s*[/\\-]?\s*){2,}/gi,
        ' '
      );


  /*
   * -------------------------------------------------------
   * 6. REMOVE PROMO / TRANSLATION JUNK
   * -------------------------------------------------------
   */
  text =
    text
      .replace(
        /\b(?:Subukan Natin|Subukan natin|Matuto Pa|Matuto pa|Alamin Pa|Alamin pa|Basahin Pa|Basahin pa|Read More|Learn More|Find Out More)\b[.!]?\s*$/gi,
        ''
      );


  /*
   * -------------------------------------------------------
   * 7. FINAL TYPOGRAPHIC CLEAN
   * -------------------------------------------------------
   */
  text =
    text
      .replace(
        /\s+-\s+/g,
        '-'
      )
      .replace(
        /\s+([,.!?;:])/g,
        '$1'
      )
      .replace(
        /\(\s*\)/g,
        ' '
      )
      .replace(
        /\s{2,}/g,
        ' '
      )
      .trim();


  /*
   * URL-only content must show NOTHING.
   */
  if (
    text.length <
      35 ||
    !/[A-Za-z\u00C0-\u00FF]/.test(
      text
    )
  ) {

    return '';

  }


  return text;

}

function HomeImportantReminder() {

  const {
    hasRole,
  } =
    useAuth();


  const [
    reminder,
    setReminder,
  ] =
    useState<
      HomeImportantReminderRow |
      null
    >(null);


  useEffect(
    () => {

      let active =
        true;


      const load =
        async () => {

          const {
            data,
            error,
          } =
            await supabase
              .from(
                'home_important_reminders'
              )
              .select(
                'slot,title,body,is_active,starts_at,ends_at,updated_at,media_path'
              )
              .eq(
                'slot',
                'home_primary'
              )
              .maybeSingle();


          if (
            !active ||
            error ||
            !data
          ) {
            return;
          }


          const now =
            Date.now();


          const visible =
            data.is_active &&
            Date.parse(
              data.starts_at
            ) <= now &&
            (
              !data.ends_at ||
              Date.parse(
                data.ends_at
              ) > now
            );


          if (!visible) {

            setReminder(
              null
            );

            return;
          }


          let mediaUrl:
            string |
            null =
            null;


          if (
            data.media_path
          ) {

            const {
              data: signed,
            } =
              await supabase
                .storage
                .from(
                  'bulletin-media'
                )
                .createSignedUrl(
                  data.media_path,
                  3600
                );


            if (!active) {
              return;
            }


            mediaUrl =
              signed?.signedUrl ??
              null;
          }


          setReminder({
            ...(data as HomeImportantReminderRow),
            media_url:
              mediaUrl,
          });

        };


      void load();


      return () => {
        active =
          false;
      };

    },
    []
  );


  const canEdit =
    hasRole(
      'system_admin'
    );


  if (
    !reminder &&
    !canEdit
  ) {
    return null;
  }


  return (

    <View
      style={{
        marginTop: 8,
      }}
    >

      <SectionTitle>
        IMPORTANT REMINDER
      </SectionTitle>


      <Card>

        {reminder ? (

          <View>

            {!!reminder.media_url && (

              <BulletinImage
                source={{
                  uri:
                    reminder.media_url,
                }}
                style={{
                  width: '100%',
                  aspectRatio: 16 / 9,
                  borderRadius: 14,
                  marginBottom: 14,
                }}
                contentFit="cover"
                transition={180}
              />

            )}


            <View
              style={{
                borderLeftWidth: 4,
                borderLeftColor:
                  colors.greenDark,
                paddingLeft: 11,
              }}
            >

              <Text
                style={{
                  fontSize: 17,
                  fontWeight: '900',
                  color:
                    colors.greenDark,
                  marginBottom: 7,
                }}
              >
                {reminder.title}
              </Text>


              <Text
                style={{
                  fontSize: 14,
                  lineHeight: 21,
                  color: '#374151',
                }}
              >
                {reminder.body}
              </Text>

            </View>

          </View>

        ) : (

          <Text
            style={
              shared.muted
            }
          >
            No active reminder.
          </Text>

        )}


        {canEdit && (

          <Pressable
            onPress={() =>
              router.push(
                '/admin/home-reminder' as any
              )
            }
            style={{
              marginTop: 14,
              flexDirection: 'row',
              alignItems: 'center',
              justifyContent:
                'center',
              gap: 6,
              paddingVertical: 9,
            }}
          >

            <Text
              style={{
                fontSize: 12,
                fontWeight: '900',
                color:
                  colors.greenDark,
              }}
            >
              EDIT IMPORTANT REMINDER
            </Text>

            <MaterialCommunityIcons
              name="pencil-outline"
              size={16}
              color={
                colors.greenDark
              }
            />

          </Pressable>

        )}

      </Card>

    </View>

  );

}


function HomeMissionBulletins() {

  const {
    profile,
  } =
    useAuth();


  const mounted =
    useBulletinRef(
      true
    );


  const [
    items,
    setItems,
  ] =
    useState<
      MissionBulletin[]
    >([]);


  const [
    loading,
    setLoading,
  ] =
    useState(
      true
    );


  const [
    engagement,
    setEngagement,
  ] =
    useState<
      Record<
        string,
        {
          love: number;
          praying: number;
          amen: number;
          mine:
            | 'love'
            | 'praying'
            | 'amen'
            | null;
        }
      >
    >({});


  const [
    comments,
    setComments,
  ] =
    useState<
      Record<
        string,
        any[]
      >
    >({});


  const [
    commentDrafts,
    setCommentDrafts,
  ] =
    useState<
      Record<
        string,
        string
      >
    >({});


  const [
    replyingTo,
    setReplyingTo,
  ] =
    useState<{
      bulletinId: string;
      commentId: string;
      name: string;
    } | null>(
      null
    );


  const [
    replyDraft,
    setReplyDraft,
  ] =
    useState('');


  const [
    busyKey,
    setBusyKey,
  ] =
    useState<
      string |
      null
    >(null);


  useEffect(
    () => {

      mounted.current =
        true;


      return () => {
        mounted.current =
          false;
      };

    },
    []
  );


  const load =
    async () => {

      if (
        mounted.current
      ) {
        setLoading(
          true
        );
      }


      const now =
        new Date()
          .toISOString();


      let query =
        supabase
          .from(
            'mission_bulletins'
          )
          .select(
            'id,church_id,title,body,priority,is_pinned,starts_at,expires_at,created_at,media_path,church:church_id(name)'
          )
          .eq(
            'is_published',
            true
          )
          .lte(
            'starts_at',
            now
          )
          .order(
            'is_pinned',
            {
              ascending:
                false,
            }
          )
          .order(
            'priority',
            {
              ascending:
                false,
            }
          )
          .order(
            'created_at',
            {
              ascending:
                false,
            }
          );


      if (
        profile?.church_id
      ) {

        query =
          query.or(
            'church_id.is.null,church_id.eq.' +
            profile.church_id
          );

      } else {

        query =
          query.is(
            'church_id',
            null
          );

      }


      const {
        data,
        error,
      } =
        await query;


      if (
        !mounted.current
      ) {
        return;
      }


      if (error) {

        console.warn(
          'Could not load Mission Bulletins:',
          error.message
        );

        setItems([]);
        setLoading(false);

        return;
      }


      const activeItems =
        (
          data ??
          []
        )
          .filter(
            (item: any) =>
              !item.expires_at ||
              Date.parse(
                item.expires_at
              ) >
                Date.now()
          );


      const withMedia =
        await Promise.all(

          activeItems.map(
            async (
              item: any
            ) => {

              if (
                !item.media_path
              ) {
                return {
                  ...item,
                  media_url:
                    null,
                };
              }


              const {
                data:
                  signed,
              } =
                await supabase
                  .storage
                  .from(
                    'bulletin-media'
                  )
                  .createSignedUrl(
                    item.media_path,
                    3600
                  );


              return {
                ...item,
                media_url:
                  signed?.signedUrl ??
                  null,
              };

            }
          )

        );


      if (
        !mounted.current
      ) {
        return;
      }


      setItems(
        withMedia as
          MissionBulletin[]
      );


      const ids =
        withMedia.map(
          (item: any) =>
            item.id
        );


      if (
        !ids.length
      ) {

        setEngagement({});
        setComments({});
        setLoading(false);

        return;
      }


      const [
        reactionResult,
        commentResult,
      ] =
        await Promise.all([

          supabase
            .from(
              'mission_bulletin_reactions'
            )
            .select(
              'bulletin_id,user_id,reaction'
            )
            .in(
              'bulletin_id',
              ids
            ),

          supabase
            .from(
              'mission_bulletin_comments'
            )
            .select(
              'id,bulletin_id,user_id,parent_id,body,created_at'
            )
            .in(
              'bulletin_id',
              ids
            )
            .order(
              'created_at',
              {
                ascending:
                  true,
              }
            ),

        ]);


      if (
        !mounted.current
      ) {
        return;
      }


      const nextEngagement:
        Record<
          string,
          {
            love: number;
            praying: number;
            amen: number;
            mine:
              | 'love'
              | 'praying'
              | 'amen'
              | null;
          }
        > =
          {};


      ids.forEach(
        (id: string) => {

          nextEngagement[id] = {
            love: 0,
            praying: 0,
            amen: 0,
            mine: null,
          };

        }
      );


      (
        reactionResult.data ??
        []
      ).forEach(
        (row: any) => {

          const bucket =
            nextEngagement[
              row.bulletin_id
            ];


          if (!bucket) {
            return;
          }


          if (
            row.reaction ===
            'love'
          ) {
            bucket.love += 1;
          }


          if (
            row.reaction ===
            'praying'
          ) {
            bucket.praying += 1;
          }


          if (
            row.reaction ===
            'amen'
          ) {
            bucket.amen += 1;
          }


          if (
            row.user_id ===
            profile?.id
          ) {

            bucket.mine =
              row.reaction;

          }

        }
      );


      const rawComments =
        commentResult.data ??
        [];


      const userIds =
        Array.from(
          new Set(
            rawComments.map(
              (row: any) =>
                row.user_id
            )
          )
        );


      const nameMap:
        Record<
          string,
          string
        > =
          {};


      if (
        userIds.length
      ) {

        const {
          data:
            profileRows,
        } =
          await supabase
            .from(
              'profiles'
            )
            .select(
              'id,full_name'
            )
            .in(
              'id',
              userIds as string[]
            );


        (
          profileRows ??
          []
        ).forEach(
          (row: any) => {

            nameMap[
              row.id
            ] =
              row.full_name ||
              'Member';

          }
        );

      }


      if (
        !mounted.current
      ) {
        return;
      }


      const nextComments:
        Record<
          string,
          any[]
        > =
          {};


      ids.forEach(
        (id: string) => {
          nextComments[id] =
            [];
        }
      );


      rawComments.forEach(
        (row: any) => {

          if (
            !nextComments[
              row.bulletin_id
            ]
          ) {
            return;
          }


          nextComments[
            row.bulletin_id
          ].push({
            ...row,
            author_name:
              nameMap[
                row.user_id
              ] ||
              (
                row.user_id ===
                profile?.id
                  ? (
                      profile as any
                    )?.full_name ||
                    'You'
                  : 'Member'
              ),
          });

        }
      );


      setEngagement(
        nextEngagement
      );

      setComments(
        nextComments
      );

      setLoading(
        false
      );

    };


  useEffect(
    () => {

      void load();

    },
    [
      profile?.church_id,
      profile?.id,
    ]
  );


  const react =
    async (
      bulletinId:
        string,
      reaction:
        'love' |
        'praying' |
        'amen'
    ) => {

      if (
        !profile?.id
      ) {
        return;
      }


      const current =
        engagement[
          bulletinId
        ]?.mine ??
        null;


      const next =
        current === reaction
          ? null
          : reaction;


      setBusyKey(
        'reaction:' +
        bulletinId
      );


      const {
        error,
      } =
        await (
          supabase as any
        ).rpc(
          'set_mission_bulletin_reaction_v1',
          {
            p_bulletin_id:
              bulletinId,
            p_reaction:
              next,
          }
        );


      if (error) {

        BulletinAlert.alert(
          'Reaction',
          error.message
        );

        setBusyKey(
          null
        );

        return;
      }


      await load();


      if (
        mounted.current
      ) {
        setBusyKey(
          null
        );
      }

    };


  const submitComment =
    async (
      bulletinId:
        string,
      parentId:
        string |
        null =
        null
    ) => {

      if (
        !profile?.id
      ) {
        return;
      }


      const text =
        parentId
          ? replyDraft.trim()
          : (
              commentDrafts[
                bulletinId
              ] ??
              ''
            ).trim();


      if (!text) {
        return;
      }


      setBusyKey(
        'comment:' +
        bulletinId
      );


      const {
        error,
      } =
        await supabase
          .from(
            'mission_bulletin_comments'
          )
          .insert({
            bulletin_id:
              bulletinId,
            user_id:
              profile.id,
            parent_id:
              parentId,
            body:
              text,
          });


      if (error) {

        BulletinAlert.alert(
          parentId
            ? 'Reply'
            : 'Comment',
          error.message
        );

        setBusyKey(
          null
        );

        return;
      }


      if (parentId) {

        setReplyDraft('');
        setReplyingTo(
          null
        );

      } else {

        setCommentDrafts(
          (previous) => ({
            ...previous,
            [bulletinId]:
              '',
          })
        );

      }


      await load();


      if (
        mounted.current
      ) {
        setBusyKey(
          null
        );
      }

    };


  const deleteComment =
    async (
      commentId:
        string
    ) => {

      const {
        error,
      } =
        await supabase
          .from(
            'mission_bulletin_comments'
          )
          .delete()
          .eq(
            'id',
            commentId
          );


      if (error) {

        BulletinAlert.alert(
          'Comment',
          error.message
        );

        return;
      }


      await load();

    };


  if (loading) {

    return (

      <Card>

        <ActivityIndicator
          color={
            colors.green
          }
        />

      </Card>

    );

  }


  if (
    !items.length
  ) {

    return (

      <Card>

        <Text
          style={
            shared.muted
          }
        >
          No Mission Bulletins yet.
        </Text>

      </Card>

    );

  }


  return (

    <View
      style={{
        gap: 4,
      }}
    >

      {items.map(
        (item) => {

          const churchName =
            Array.isArray(
              item.church
            )
              ? item.church[
                  0
                ]?.name
              : item.church
                  ?.name;


          const reaction =
            engagement[
              item.id
            ] ?? {
              love: 0,
              praying: 0,
              amen: 0,
              mine: null,
            };


          const allComments =
            comments[
              item.id
            ] ??
            [];


          const rootComments =
            allComments.filter(
              (comment) =>
                !comment.parent_id
            );


          return (

            <Card
              key={
                item.id
              }
            >

              {!!item.media_url && (

                <BulletinImage
                  source={{
                    uri:
                      item.media_url,
                  }}
                  style={{
                    width:
                      '100%',
                    aspectRatio:
                      16 / 9,
                    borderRadius:
                      14,
                    marginBottom:
                      14,
                  }}
                  contentFit="cover"
                  transition={180}
                />

              )}


              <View
                style={{
                  borderLeftWidth:
                    5,
                  borderLeftColor:
                    colors.green,
                  paddingLeft:
                    12,
                }}
              >

                <Text
                  style={{
                    color:
                      colors.green,
                    fontWeight:
                      '900',
                    fontSize: 11,
                    letterSpacing:
                      1,
                  }}
                >

                  {item.is_pinned
                    ? 'PINNED | '
                    : ''}

                  {churchName ||
                    'ALL LOCALS'}

                </Text>


                <Text
                  style={{
                    marginTop: 7,
                    fontSize: 18,
                    lineHeight: 23,
                    fontWeight:
                      '900',
                    color:
                      '#111827',
                  }}
                >
                  {item.title}
                </Text>


                <Text
                  style={{
                    marginTop: 8,
                    fontSize: 14,
                    lineHeight: 21,
                    color:
                      '#374151',
                  }}
                >
                  {item.body}
                </Text>


                <Text
                  style={{
                    marginTop: 9,
                    fontSize: 11,
                    color:
                      '#6B7280',
                  }}
                >
                  {new Date(
                    item.starts_at
                  ).toLocaleString()}
                </Text>

              </View>


              {/* REACTIONS */}

              <View
                style={{
                  flexDirection:
                    'row',
                  gap: 7,
                  marginTop: 14,
                }}
              >

                {([
                  [
                    'love',
                    '\u2764\uFE0F',
                    'Love',
                    reaction.love,
                  ],
                  [
                    'praying',
                    '\uD83D\uDE4F',
                    'Praying',
                    reaction.praying,
                  ],
                  [
                    'amen',
                    '\uD83D\uDE4C',
                    'Amen',
                    reaction.amen,
                  ],
                ] as const)
                  .map(
                    (
                      [
                        key,
                        icon,
                        label,
                        count,
                      ]
                    ) => {

                      const selected =
                        reaction.mine ===
                        key;


                      return (

                        <Pressable
                          key={
                            key
                          }
                          disabled={
                            busyKey ===
                            'reaction:' +
                              item.id
                          }
                          onPress={() =>
                            void react(
                              item.id,
                              key
                            )
                          }
                          style={{
                            flex: 1,
                            minHeight:
                              40,
                            borderRadius:
                              12,
                            borderWidth:
                              1,
                            borderColor:
                              selected
                                ? colors.green
                                : '#E5E7EB',
                            backgroundColor:
                              selected
                                ? '#ECFDF3'
                                : '#FFFFFF',
                            alignItems:
                              'center',
                            justifyContent:
                              'center',
                            paddingHorizontal:
                              4,
                          }}
                        >

                          <Text
                            style={{
                              fontSize:
                                12,
                              fontWeight:
                                selected
                                  ? '900'
                                  : '700',
                              color:
                                selected
                                  ? colors.greenDark
                                  : '#4B5563',
                            }}
                          >
                            {icon}{' '}
                            {label}{' '}
                            {count}
                          </Text>

                        </Pressable>

                      );

                    }
                  )}

              </View>


              {/* COMMENTS */}

              <View
                style={{
                  marginTop:
                    15,
                  borderTopWidth:
                    1,
                  borderTopColor:
                    '#E5E7EB',
                  paddingTop:
                    12,
                }}
              >

                <Text
                  style={{
                    fontSize: 13,
                    fontWeight:
                      '900',
                    color:
                      colors.greenDark,
                    marginBottom:
                      9,
                  }}
                >
                  COMMENTS ({allComments.length})
                </Text>


                {rootComments.map(
                  (
                    comment
                  ) => {

                    const replies =
                      allComments.filter(
                        (
                          child
                        ) =>
                          child.parent_id ===
                          comment.id

  reading_date: string;
  plan_day_number: number;
  plan_week_number: number;
  cycle_number: number;
  chapters_today: number;

  start_book_no: number;
  start_book_name: string;
  start_chapter: number;

  end_book_no: number;
  end_book_name: string;
  end_chapter: number;

  bible_reference: string;
};


function HomeDailyBibleReminder() {

  const [
    plan,
    setPlan,
  ] =
    React.useState<HomeBiblePlan | null>(
      null
    );


  const [
    loading,
    setLoading,
  ] =
    React.useState(true);


  React.useEffect(() => {

    let cancelled =
      false;


    const loadToday =
      async () => {

        try {

          const result =
            await (supabase as any).rpc(
              'get_automatic_bible_today'
            );


          if (
            cancelled
          ) {
            return;
          }


          if (
            result?.error
          ) {

            /*
             * Graceful fallback while automatic
             * Bible migrations are not deployed.
             */
            console.warn(
              'Home Daily Bible:',
              result.error?.message
            );

            setPlan(null);
            return;
          }


          const raw =
            Array.isArray(
              result?.data
            )
              ? result.data[0]
              : result?.data;


          if (
            raw &&
            raw.start_book_no &&
            raw.start_chapter &&
            raw.reading_date
          ) {

            setPlan({
              reading_date:
                String(
                  raw.reading_date
                ),

              plan_day_number:
                Number(
                  raw.plan_day_number ??
                  0
                ),

              plan_week_number:
                Number(
                  raw.plan_week_number ??
                  0
                ),

              cycle_number:
                Number(
                  raw.cycle_number ??
                  1
                ),

                Number(
                  raw.chapters_today ??
                  1
                ),

                Number(
                  raw.start_book_no
                ),

              start_book_name:
                String(
                  raw.start_book_name ??
                  ''
                ),

              start_chapter:
                Number(
                  raw.start_chapter
                ),

              end_book_no:
                Number(
                  raw.end_book_no ??
                  raw.start_book_no
                ),

              end_book_name:
                String(
                  raw.end_book_name ??
                  raw.start_book_name ??
                  ''
                ),

              end_chapter:
                Number(
                  raw.end_chapter ??
                  raw.start_chapter
                ),

              bible_reference:
                String(
                  raw.bible_reference ??
                  ''
                ),
            });

          } else {

            setPlan(null);
          }

        } catch (
          e: any

          if (
            !cancelled

            console.warn(
              'Home Daily Bible unavailable:',
              e?.message ?? e
            );

            setPlan(null);
          }

        } finally {

          if (
            !cancelled

            setLoading(false);
          }
        }
      };


    loadToday();


    return () => {
      cancelled = true;
    };

  }, []);


  const openReading =

              String(

              ),

              'daily_plan',

            assignedDate:
              plan.reading_date,
          },
        });

        return;
      }

       */
      router.push(
        '/bible'
      );
    };

            flex: 1,

                0.6,

                4,
            }}
          >
            TODAY'S DAILY BIBLE
          </Text>


          <Text
            style={{
              fontSize: 15,

              fontWeight:
                '900',

              color:
                colors.greenDark,
            }}
          >
            {
              loading
                ? 'Loading today\'s reading...'
                : plan?.bible_reference ||
                  'Daily Bible Reading'
            }
          </Text>


          <Text

function CommunityZoomPhoto({
 uri
}:{
 uri:string
}){
 const scale=
  useSharedValue(1);

 const startScale=
  useSharedValue(1);


 const pinch=
  Gesture.Pinch()

   .onStart(()=>{
    startScale.value=
     scale.value;
   })

   .onUpdate(event=>{
    scale.value=
     Math.max(
      1,
      Math.min(5,
       startScale.value*
       event.scale
      )
     );
   })

   .onEnd(()=>{
    if(scale.value<1.05){
     scale.value=
      withTiming(1);
    }
   });


 const doubleTap=
  Gesture.Tap()

   .numberOfTaps(2)

   .maxDuration(280)

   .onEnd(
    (_event,success)=>{
     if(!success){
      return;
     }

     scale.value=
      withTiming(
       scale.value>1.1
        ? 1
        : 3
      );
    }
   );


 const gesture=
  Gesture.Simultaneous(
   pinch,
   doubleTap
  );


 const animatedStyle=
  useAnimatedStyle(
   ()=>({
    transform:[
     {
      scale:
       scale.value
     }
    ]
   })
  );


 return (
  <GestureDetector
   gesture={gesture}
  >
   <Animated.View
    collapsable={false}
    style={{
     width:'100%',
     height:'100%',
     alignItems:'center',
     justifyContent:'center'
    }}
   >
    {/* PATCH_161H_D1_ZOOM_CONTAINER */}
    <Animated.Image

    ]}

 );

 const prepareCommunityPhoto =
  async(asset:any)=>{
   const M=
    await import(
     'expo-image-manipulator'
    );

   const actions:any[]=[];

   if(
    Number(asset.width??0)>
    1400
   ){
    actions.push({
     resize:{width:1400}
    });
   }

   const x=
    await M.manipulateAsync(
     asset.uri,
     actions,
     {
      compress:.78,
      format:
       M.SaveFormat.JPEG,
      base64:true
     }
    );

   if(!x.base64){
    throw new Error(
     'Could not prepare image.'
    );
   }

   const sizeBytes=
    Math.ceil(
     x.base64.length*.75
    );

   if(
    sizeBytes >
    5*1024*1024
   ){
    throw new Error(
     'Photo is too large.'
    );
   }

   setPhotos(current=>
       current.length>=5
        ? current
        : [
           ...current,
           {
            
    uri:x.uri,
    base64:x.base64!,
    sizeBytes
   
           }
          ].slice(0,5)
      );
  };

   }>=[];

  try{
   for(

   ){
    const photo=
     photos[index];

     await supabase

      )
      .upload(
       path,
       decode(
        photo.base64
       ),
       {
        contentType:
         'image/jpeg',
        upsert:false
       }
      );

    if(uploadError){
     throw uploadError;
    }

    uploaded.push({
     media_path:path,
     media_mime_type:
      'image/jpeg',
     media_size_bytes:
      photo.sizeBytes
    });
   }

   const first=
    uploaded[0]??null;

   const {
    data:created,
    error:createError
   }=
    await (
     supabase as any
    ).rpc(
     'create_buklod_community_post_v3',
     {
      p_audience:
       audience,
      p_group_id:
       audience==='group'
        ? group?.id??null
        : null,
      p_category:
       category,
      p_body:
       body.trim()||
       (
        uploaded.length>1
         ? 'Shared photos.'
         : 'Shared a photo.'
       ),
      p_media_path:
       first?.media_path??
       null,
      p_media_mime_type:
       first?.media_mime_type??
       null,
      p_media_size_bytes:
       first?.media_size_bytes??
       null,
      p_mentioned_user_ids:
       mentions.map(
        x=>x.user_id
       )
     }
    );

   if(createError){
    throw createError;
   }

   createdPostId=
    typeof created==='string'
     ? created
     : Array.isArray(created)
     ? (
        typeof created[0]==='string'
         ? created[0]
         : created[0]?.id??null
       )
     : created?.id??null;

   if(!createdPostId){
    throw new Error(
     'Post was created but its ID was not returned.'
    );
   }

   if(uploaded.length){
    const {
     error:mediaError
    }=
     await (
      supabase as any
     ).rpc(

 const react=async(
  postId:string,
  reaction:Reaction
)=>{
  /* PATCH_154N_OPTIMISTIC_COMMUNITY_REACTION */

  const current=
    posts.find(
      p=>p.id===postId
    );

  const previous=
    (
      current?.my_reaction??
      null
    ) as Reaction|null;

  const nextReaction=
    previous===reaction
      ? null
      : reaction;

  const changeReaction=(
    base:any,
    from:Reaction|null,
    to:Reaction|null
  )=>{
    const counts={
      ...(
        base.reaction_counts||
        {}
      )
    };

    let total=
      Number(
        base.total_reactions??
        Object.values(
          counts
        ).reduce(
          (
            sum:number,
            value:any
          )=>
            sum+
            Number(
              value??0
            ),
          0
        )
      );

    if(from){
      counts[from]=
        Math.max(
          0,
          Number(
            counts[from]??0
          )-1
        );

      total=
        Math.max(
          0,
          total-1
        );
    }

    if(to){
      counts[to]=
        Number(
          counts[to]??0
        )+1;

      total+=1;
    }

    return{
      ...base,
      reaction_counts:
        counts,
      total_reactions:
        total,
      my_reaction:
        to
    };
  };

  /* Count changes NOW */
  setPosts(
    currentPosts=>
      currentPosts.map(
        p=>
          p.id===postId
            ? changeReaction(
                p,
                previous,
                nextReaction
              )
            : p
      )
  );

  const {error}=
    await(
      supabase as any
    ).rpc(
      'set_buklod_post_reaction_v3',
      {
        p_post_id:
          postId,
        p_reaction:
          reaction
      }
    );

  if(error){
    /* DB rejected: restore count */
    setPosts(
      currentPosts=>
        currentPosts.map(
          p=>
            p.id===postId
              ? changeReaction(
                  p,
                  nextReaction,
                  previous
                )
              : p
        )
    );

    Alert.alert(
      'Reaction',
      error.message
    );
  }
};

 const reactToComment=
  async(
   reaction:Reaction
  )=>{
   const target=
    commentReactionTarget;

   if(!target?.id){
    return;
   }

   const targetId=
    String(target.id);

   const previous=
    (
     target.my_reaction??
     null
    ) as
     Reaction|null;

   const nextReaction=
    previous===reaction
     ? null
     : reaction;

   setCommentReactionOpen(
    false
   );

   setCommentReactionTarget(
    null
   );

   setComments(
    current=>
     current.map(
      item=>{
       if(
        String(item.id)!==
        targetId
       ){
        return item;
       }

       const counts={
        ...(
         item.reaction_counts??
         {}
        )
       };

       if(previous){
        counts[previous]=
         Math.max(
          0,
          Number(
           counts[previous]??
           0
          )-1
         );
       }

       if(nextReaction){
        counts[nextReaction]=
         Number(
          counts[nextReaction]??
          0
         )+1;
       }

       return {
        ...item,
        my_reaction:
         nextReaction,
        reaction_counts:
         counts
       };
      }
     )
   );

   const {error}=
    await (
     supabase as any
    ).rpc(
     'set_buklod_comment_reaction_v1',
     {
      p_comment_id:
       targetId,
      p_reaction:
       reaction
     }
    );

   if(error){
    Alert.alert(
     'Reaction',
     error.message
    );

    if(commentPost?.id){
     await loadComments(
      commentPost.id
     );
    }
   }
  };
