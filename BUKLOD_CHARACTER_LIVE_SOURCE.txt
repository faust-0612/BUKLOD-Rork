type Props = {
  lifetimePoints?: number;
  rankName?: string | null;
};

/** Profile card â€” MY BUKLOD CHARACTER with ranking, points, and one Customize. */
export default function BuklodProfileCharacterCard({
  lifetimePoints = 0,
  rankName = null,
}: Props) {
  const { theme } = useBuklodPresentation();
  const { session } = useAuth();
  const [focused, setFocused] = useState(false);
  const characterEvents = useBuklodCharacterEventsOptional();
  const greetedRef = useRef(false);
  const userId = session?.user?.id ?? null;
  const [config, setConfig] =
    useState<BuklodCharacterState>({
      ...DEFAULT_CHARACTER,
    });
  const [customizeOpen, setCustomizeOpen] =
    useState(false);

  useEffect(() => {
    void loadBuklodCharacter(userId).then((loaded) => {
      const reconciled = reconcileCharacterForPoints(loaded, lifetimePoints);
      setConfig(reconciled);
    });
  }, [lifetimePoints, userId]);

  useFocusEffect(
    useCallback(() => {
      setFocused(true);
      return () => setFocused(false);
    }, [])
  );

  useEffect(() => {
    if (!focused || greetedRef.current) return;
    greetedRef.current = true;
    characterEvents?.emit({
      type: CharacterEventType.PROFILE_CHARACTER_VISIBLE,
      success: true,
    });
  }, [characterEvents, focused]);

  const rankLabel = String(rankName ?? '').trim();

  const styles = StyleSheet.create({
    wrap: {
      marginHorizontal: 12,
      marginBottom: 6,
    },
    card: {
      padding: 14,
      borderRadius: 16,
      borderWidth: 1,
      borderColor: theme.border,
      backgroundColor: withAlpha(theme.cardSurface, 0.82),
    },
    eyebrow: {
      fontSize: 9,
      fontWeight: '900',
      letterSpacing: 1.2,
      color: theme.textMuted,
      marginBottom: 10,
    },
    body: {
      flexDirection: 'row',
      alignItems: 'center',
      gap: 12,
    },
    avatarPress: {
      alignItems: 'center',
      justifyContent: 'center',
    },
    copyCol: {
      flex: 1,
      minWidth: 0,
      justifyContent: 'center',
      gap: 8,
    },
    rankTitle: {
      fontSize: 15,
      fontWeight: '900',
      color: theme.textPrimary,
    },
    pointsMeta: {
      marginTop: 2,
      fontSize: 12,
      fontWeight: '800',
      color: theme.textSecondary,
    },
    customize: {
      alignSelf: 'flex-start',
      minHeight: 36,
      paddingHorizontal: 12,
      borderRadius: 12,
      flexDirection: 'row',
      alignItems: 'center',
      justifyContent: 'center',
      gap: 6,
      backgroundColor: theme.accentSoft,
      borderWidth: 1,
      borderColor: theme.border,
    },
    customizeText: {
      fontSize: 11,
      fontWeight: '900',
      color: theme.accent,
    },
  });

  return (
    <>
      <ProfileMotionFadeUp delay={55}>
        <View style={styles.wrap}>
          <View style={styles.card}>
            <Text style={styles.eyebrow}>
              MY BUKLOD CHARACTER
            </Text>

            <View style={styles.body}>
              <Pressable
                accessibilityRole="button"
                accessibilityLabel="Open ranking and BUKLOD Points"
                onPress={() => router.push('/points')}
                style={({ pressed }) => [
                  styles.avatarPress,
                  pressed && { opacity: 0.92 },
                ]}
              >
                /* BUKLOD_CHARACTER_ENGINE_2_NATIVE_SKIA_R1 */
/* BUKLOD_PROFILE_FULL_BODY_R2 */
<BuklodCharacterCompactFullBody
  body={config.body}
  equipment={config.equipment}
  size={88}
  
/>
              </Pressable>

              <View style={styles.copyCol}>
                <Pressable
                  accessibilityRole="button"
                  accessibilityLabel="Open ranking and BUKLOD Points"
                  onPress={() => router.push('/points')}
                >
                  <Text style={styles.rankTitle} numberOfLines={1}>
                    {rankLabel || 'BUKLOD Points'}
                  </Text>
                  <Text style={styles.pointsMeta}>
                    {formatPoints(lifetimePoints)} pts
                  </Text>
                </Pressable>

                <Pressable
                  onPress={() => setCustomizeOpen(true)}
                  accessibilityRole="button"
                  accessibilityLabel="Customize character"
                  style={({ pressed }) => [
                    styles.customize,
                    pressed && { opacity: 0.88 },
                  ]}
                >
                  <MaterialCommunityIcons
                    name="palette-outline"
                    size={16}
                    color={theme.accent}
                  />
                  <Text style={styles.customizeText}>
                    Customize
                  </Text>
                </Pressable>
              </View>
            </View>
          </View>
        </View>
      </ProfileMotionFadeUp>

      <BuklodCharacterCustomizeSheet
        visible={customizeOpen}
        onClose={() => setCustomizeOpen(false)}
        userId={userId}
        lifetimePoints={lifetimePoints}
        onSaved={setConfig}
      />
    </>
  );
}

function equippedId(
  equipment: BuklodCharacterState['equipment'],
  slot: SlotId
): string | null {
  const raw = equipment[slot];
  if (slot === 'accessory' && !raw) return 'acc_none';
  return raw;
}
